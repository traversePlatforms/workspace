# 365aeration.com is the main address (2026-10-14). The old aeration365.com forwards to it, keeping the path.
# Let's Encrypt renewals (.well-known) are left alone on both names.
# This file is the LIVE copy of ~/www/365aeration.com/public_html/.htaccess (a365 account; 365aeration.com is a symlink to
# the aeration365.com folder, one file serves both names). Brought back in step 2026-11-09 (overnight review): it had fallen
# behind the server since 25 Oct. Edit here, upload with a backup, flush the cache.
RewriteEngine On
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTP_HOST} ^(www\.)?aeration365\.com$ [NC]
RewriteRule ^ https://365aeration.com%{REQUEST_URI} [L,R=301]
# 2026-11-09 (overnight review): www.365aeration.com answered 200 with the same page, a second address for the one site; the
# canonical tag, the sitemap and every printed QR code say the bare name.
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTP_HOST} ^www\.365aeration\.com$ [NC]
RewriteRule ^ https://365aeration.com%{REQUEST_URI} [L,R=301]
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
